32/** \brief Summarize request lookup timings and the PHP request's peak memory. */
33function test_server_performance(array $nanoseconds, int|float $elapsed, int $peakMemory): string {
34 sort($nanoseconds, SORT_NUMERIC);
35 $samples = count($nanoseconds);
36 if ($samples === 0) { return ''; }
37 $middle = intdiv($samples, 2);
38 $median = $samples % 2 ? $nanoseconds[$middle] : ($nanoseconds[$middle - 1] + $nanoseconds[$middle]) / 2;
39 $p95 = $nanoseconds[(int)ceil($samples * 0.95) - 1];
40 return '<section><h2>Performance</h2><ul>'.
41 '<li>Lookups measured: '.$samples.'</li>'.
42 sprintf('<li>Total test time: %.3f s</li>', $elapsed / 1e9).
43 sprintf('<li>Average lookup: %.3f ms</li>', array_sum($nanoseconds) / $samples / 1e6).
44 sprintf('<li>Median lookup: %.3f ms</li>', $median / 1e6).
45 sprintf('<li>95th-percentile lookup: %.3f ms</li>', $p95 / 1e6).
46 sprintf('<li>Slowest lookup: %.3f ms</li>', $nanoseconds[$samples - 1] / 1e6).
47 sprintf('<li>PHP request peak memory: %.2f MiB</li>', $peakMemory / 1048576).
48 '</ul><p>Lookup timings include configuration loading, authentication, the database connection, and the query. '.
49 'They exclude HTTP transport and browser rendering. Peak memory includes this PHP request and its test results; '.
50 'database server memory is outside this measurement.</p></section>';
51}
