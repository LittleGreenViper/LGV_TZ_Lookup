33/** \brief Read the server's existing database settings without starting the HTTP endpoint. */
34function updateSettings(string $config): array {
35 if (!is_file($config) || !is_readable($config)) { throw new RuntimeException('The private configuration file is not readable.'); }
36 require $config;
37 if (!isset($g_dbName, $g_dbUserName, $g_dbPassword) || !is_string($g_dbName) || $g_dbName === '') {
38 throw new RuntimeException('The configuration must define g_dbName, g_dbUserName, and g_dbPassword.');
39 }
40 $driver = strtolower($g_dbType ?? 'mysql');
41 if (!in_array($driver, ['mysql', 'pgsql'], true)) { throw new RuntimeException('The database driver must be mysql or pgsql.'); }
42 return ['database' => $g_dbName, 'user' => $g_dbUserName, 'password' => $g_dbPassword,
43 'driver' => $driver, 'host' => $g_dbHost ?? '127.0.0.1', 'port' => $g_dbPort ?? ($driver === 'pgsql' ? 5432 : 3306)];
44}

47/** \brief Connect and hold the same database lock used by the installer. */
48function updateConnection(array $settings): PDO {
49 $driver = $settings['driver'];
50 $dsn = $driver.':host='.$settings['host'].';port='.$settings['port'].';dbname='.$settings['database'];
51 $dsn .= $driver === 'mysql' ? ';charset=utf8mb4' : ';options=--client_encoding=UTF8';
52 $connection = new PDO($dsn, $settings['user'], $settings['password'], [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]);
53 $sql = $driver === 'pgsql' ? "SELECT pg_try_advisory_lock(hashtext('LGV_TZ_Lookup.install'))"
54 : "SELECT GET_LOCK(CONCAT('lgv_tz_', SHA2(DATABASE(), 224)), 0)";
55 if (!$connection->query($sql)->fetchColumn()) { throw new RuntimeException('Another timezone installation or update is using this database.'); }
56 return $connection;
57}

60/** \brief Remove downloaded data and a recorded staging table after an interrupted load. */
61function updateCleanup(string $work): void {
62 foreach (['boundaries.zip', 'boundaries.json', 'release.json'] as $file) {
63 if (is_file($work.'/'.$file) && !unlink($work.'/'.$file)) { throw new RuntimeException('Could not remove downloaded '.$file); }
64 }
65 if (!is_file($work.'/update.json')) { return; }
66 $state = json_decode(file_get_contents($work.'/update.json'), true, 512, JSON_THROW_ON_ERROR);
67 if (!is_string($state['table'] ?? null) || !preg_match('/^lgv_tz_update_[a-f0-9]{16}$/D', $state['table'])) {
68 throw new RuntimeException('Invalid staging table record; refusing database cleanup.');
69 }
70 $connection = updateConnection(updateSettings($state['config']));
71 $connection->exec('DROP TABLE IF EXISTS '.$state['table']);
72 $connection->exec('DROP TABLE IF EXISTS lgv_tz_old_'.substr($state['table'], -16));
73 if (!unlink($work.'/update.json')) { throw new RuntimeException('Could not remove staging recovery information.'); }
74}

77/** \brief Load into staging, then publish atomically. The caller retains the database lock throughout. */
78function updateLoad(PDO $connection, array $settings, string $json, string $table): int {
79 if (!preg_match('/^lgv_tz_update_[a-f0-9]{16}$/D', $table)) { throw new InvalidArgumentException('Invalid staging table name.'); }
80 $database = new class($settings, $table) extends LGV_TZ_Lookup_Database {
81 /** \brief Use the normal loader schema and storage format, with this run's private destination. */
82 public function __construct(array $settings, string $table) {
83 parent::__construct($settings['database'], $settings['user'], $settings['password'],
84 $settings['driver'], $settings['host'], $settings['port']);
85 $this->_load_table = $table;
86 }
87 };
88 $stream = fopen($json, 'rb');
89 if ($stream === false) { throw new RuntimeException('Could not open the downloaded boundary file.'); }
90 try {
91 $listener = new class($database) extends LGV_TZ_Lookup_Loader {
92 public bool $complete = false; ///< True only after the parser closes the entire JSON document.
93 /** \brief Reject truncated input even when the streaming parser silently reaches EOF. */
94 public function endDocument(): void { parent::endDocument(); $this->complete = true; }
95 };
96 (new JsonStreamingParser\Parser($stream, $listener))->parse();
97 if (!$listener->complete) { throw new RuntimeException('The boundary JSON is incomplete; existing boundaries were retained.'); }
98 } finally { fclose($stream); LGV_TZ_Lookup_Loader::$db_object = null; }
99 $count = (int)$connection->query('SELECT COUNT(*) FROM '.$table)->fetchColumn();
100 if ($count === 0) { throw new RuntimeException('No polygons were loaded; existing boundaries were retained.'); }
101 if ($settings['driver'] === 'pgsql') { $connection->exec('ANALYZE '.$table); }
102 $old = 'lgv_tz_old_'.substr($table, -16);
103 if ($settings['driver'] === 'pgsql') {
104 $connection->beginTransaction();
105 try {
106 $connection->exec('ALTER TABLE timezones RENAME TO '.$old);
107 $connection->exec('ALTER TABLE '.$table.' RENAME TO timezones');
108 $connection->exec('DROP TABLE '.$old);
109 $connection->commit();
110 } catch (Throwable $error) {
111 if ($connection->inTransaction()) { $connection->rollBack(); }
112 throw $error;
113 }
114 } else {
115 $connection->exec('RENAME TABLE timezones TO '.$old.', '.$table.' TO timezones');
116 // Publication already succeeded. A failure removing the old copy must not be described as a failed load.
117 try { $connection->exec('DROP TABLE '.$old); }
118 catch (Throwable $error) { fwrite(STDERR, 'Updated boundaries, but could not remove old table '.$old.'.'.PHP_EOL); }
119 }
120 return $count;
121}

140 if (is_file($autoload)) { require_once $autoload; }
