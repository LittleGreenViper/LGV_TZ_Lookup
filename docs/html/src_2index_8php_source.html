25/**
26 \brief This is the main entrypoint for the TZ Lookup Utility.
27
28 It needs to import a configuration file, with database information.
29
30 The configuration file needs to declare the following:
31
32 $g_dbName = "<DATABASE NAME>";
33 $g_dbUserName = "<DATABASE USERNAME>";
34 $g_dbPassword = "<DATABASE USER PASSWORD>";
35 $g_dbType = "<DATABASE TYPE>";
36 $g_dbHost = "<DATABASE HOST>";
37 $g_dbPort = "<DATABASE PORT>";
38
39 If this is left blank, or omitted, then anyone can access the server (may not be good). Otherwise, they need to include "&secret=<SERVER SECRET>", when calling.
40
41 $g_server_secret = "<SERVER SECRET>";
42
43 The server can be told to re-initialize the database, but that should only be done via command line.
44
45 Otherwise, a simple long/lat HTTP query will result in a string, containing the time zone that has that location.
46*/

71function call_server( $inQuery, ///< This is an ampersand-concatenated list of parameters, either from the query, or from the arguments.
72 $inIsCLI ///< This is true, if we are calling this from the command-line.
73 ) {
74 include __CONFIG_FILE_;
75 $ret = "";

77 if (!empty($inQuery)) {
78 $queries = [];

80 if (!empty($inQuery)) {
81 $queryArray = explode('&', $inQuery);
82
83 foreach ($queryArray as $param) {
84 // Now, see if we have a bunch of parameters.
85 $key = trim($param);
86 $value = NULL;

88 $parts = explode('=', $param, 2);

104 if (!empty($queries)) {
105 $authorized = !isset($g_server_secret) || $g_server_secret === '' ||

138 // Adjust for overflow.

152 }
153 }
154 } elseif (isset($g_server_secret)) {
155 $ret = 'Ah - Ah - Aaaah! You didn\'t say the magic word!';

157 header('HTTP/1.1 403 Not Authorized');

176 $ret = "Usage: \$> php ".__FILE__." load\n\n\tInitializes the database from the GeoJSON file.\n\tIt will return 1, if successful, 0, if not. It may take some time.$secret_note\n";

Provide the requested long/lat, as comma-delimited floating-point numbers, between -180 and 180.\n\tThe response will be <a href='https://en.wikipedia.org/wiki/List_of_tz_database_time_zones'>the TZ name</a> of the timezone that contains that point.
