81 /**
82 This is the action that is called by the JSON streamer.
83 */
84 public static function listener_action( $inItem ///< The element of the JSON file being parsed. This is one geographic area for the time zone.
85 ) {
86 $tzid = $inItem["properties"]["tzid"];
87 $geometry = $inItem["geometry"]["coordinates"];
88 if ("MultiPolygon" == $inItem["geometry"]["type"]) {
89 foreach ($geometry as $coords) {
90 self::_process_entity(self::_extract_entity($tzid, $coords[0]));
91 }
92 } elseif ("Polygon" == $inItem["geometry"]["type"]) {
93 self::_process_entity(self::_extract_entity($tzid, $geometry[0]));
94 }
95 }
96
97 /***********************************************************************************************************************/
98 /**
99 This processes one new entity, after we have parsed it.
100 */
101 private static function _process_entity($inEntity ///< The new entity to be processed.
102 ) {
103 self::$db_object->store_entity($inEntity);
104 }
105
106 /***********************************************************************************************************************/
107 /**
108 This function simply compares the given long ([0], and lat ([1]) against the current values, and extends an edge, if necessary.
109
110 This is designed to be a reduce handler, with the current value, and the new value, passed in, returning a new current value.
111
112 \returns a new domain rect, as an associative array.
113 */
114 private static function _update_coords( $current, ///< This is the current domain rect. It contains the maximums for East, West, North, and South, in an associative array.
115 $next ///< This is the next value of the long lat. It is a simple 2-element floating point array, with [0] being the longitude, and [1] being the latitude.
116 ) {
117 $current['east'] = max($current['east'], $next[0]);
118 $current['west'] = min($current['west'], $next[0]);
119 $current['north'] = max($current['north'], $next[1]);
120 $current['south'] = min($current['south'], $next[1]);
121 return $current;
122 }
123
124 /***********************************************************************************************************************/
125 /**
126 This function generates a new entity, based on the given time zone ID, and the polygon array.
127
128 \returns: A new instance of LGV_TZ_Lookup_Entity.
129 */
130 private static function _extract_entity($tzid, ///< This is [the standard TZ time zone designator](https://en.wikipedia.org/wiki/List_of_tz_database_time_zones) that applies to this entity.
131 $coords_array ///< This is the exact polygon for this entitye.
132 ) {
133 $domainRect = array_reduce($coords_array, 'LGV_TZ_Lookup_Loader::_update_coords', ['east' => -1000, 'west' => 1000, 'north' => -1000, 'south' => 1000]);
134
135 return new LGV_TZ_Lookup_Entity($tzid, $domainRect, $coords_array);
136 }
