21/**
22 \brief This fixes generated alphabetical-index links and verifies the local HTML before publication.
23
24 Doxygen 1.18 emits raw dollar/underscore index links for PHP names, while the corresponding anchors are escaped.
25 Some short, ungrouped function lists also receive an alphabetical bar without section anchors. We repair only
26 these generated index links, then check all generated local files, images, scripts, styles, and HTML anchors.
27 A failed check prevents generate-docs.sh from replacing the existing documentation.
28*/
29declare(strict_types = 1);
30
31if ($argc !== 2 || !is_dir($argv[1])) {
32 fwrite(STDERR, "Expected the generated HTML directory.\n");
33 exit(1);
34}
35$directory = realpath($argv[1]);
36$pages = [];
37foreach (glob($directory.'/*.html') as $file) {
38 $html = file_get_contents($file);
39 foreach (['#index_$' => '#index__24', '#index__' => '#index__5F'] as $old => $new) {
40 if (str_contains($html, 'id="'.substr($new, 1).'"')) {
41 $html = str_replace('href="'.$old.'"', 'href="'.$new.'"', $html);
42 }
43 }
44 if (str_contains($html, 'href="#index_') && !str_contains($html, 'id="index_')) {
45 $html = preg_replace('/^<div class="qindex">[^\r\n]*<\/div>\r?\n/m', '', $html);
46 }
47 file_put_contents($file, $html);
48 preg_match_all('/\b(?:id|name)="([^"]*)"/', $html, $ids);
49 $pages[$file] = ['html' => $html, 'ids' => array_fill_keys($ids[1], true)];
50}
51$errors = [];
52foreach ($pages as $file => $page) {
53 preg_match_all('/\b(?:href|src|data)="([^"]*)"/', $page['html'], $links);
54 foreach ($links[1] as $link) {
55 $link = html_entity_decode($link, ENT_QUOTES | ENT_HTML5, 'UTF-8');
56 if ($link === '' || preg_match('/^(?:[a-z][a-z0-9+.-]*:|\/\/)/i', $link)) { continue; }
57 $url = parse_url($link);
58 $path = rawurldecode($url['path'] ?? '');
59 $target = $path === '' ? $file : realpath(dirname($file).'/'.$path);
60 if ($target === false || !file_exists($target)) {
61 $errors[] = basename($file).': missing local resource '.$link;
62 } elseif (isset($url['fragment'], $pages[$target]) &&
63 !isset($pages[$target]['ids'][rawurldecode($url['fragment'])])) {
64 $errors[] = basename($file).': missing anchor '.$link;
65 }
66 }
67}
68if (!empty($errors)) {
69 fwrite(STDERR, implode(PHP_EOL, array_unique($errors)).PHP_EOL);
70 exit(1);
71}
72echo 'Verified '.count($pages)." HTML pages and their local links/assets.\n";
