57 /**
58 This queries the database, and returns the TZ.
59
60 \returns: A string, with the timezone. Empty, if none.
61 */
62 public function get_tz( $in_lng, ///< The longitude
63 $in_lat ///< The latitude
64 ) {
65 // This does a fast lookup, using the domain rect (the "blunt instrument" rect that we created, when we stored the polygon).
66 $tzIDs = $this->db_object->get_tz_ids($in_lng, $in_lat);
67
68 $named = [];
69 $ocean = [];
70 foreach ($tzIDs as $candidate) {
71 if (str_starts_with($candidate['tzname'], 'Etc/')) {
72 $ocean[] = $candidate;
73 } else {
74 $named[] = $candidate;
75 }
76 }
77
78 // Preserve the existing single-candidate shortcut and named-zone precedence.
79 if (1 == count($named)) {
80 return $named[0]['tzname'];
81 }
82 $timezone = $this->_find_in_polygons($named, $in_lng, $in_lat);
83 if ('' !== $timezone) {
84 return $timezone;
85 }
86 if (1 == count($tzIDs)) {
87 return $tzIDs[0]['tzname'];
88 }
89
90 // Named polygons have already failed; only fetch the ocean polygons now.
91 return $this->_find_in_polygons($ocean, $in_lng, $in_lat);
92 }
93
94 /***********************************************************************************************************************/
95 /**
96 Fetch compact polygons and stop at the first match, without expanding all candidates into PHP point arrays.
97 */
98 private function _find_in_polygons($candidates, $longitude, $latitude) {
99 if (empty($candidates)) {
100 return '';
101 }
102 $ids = array_column($candidates, 'id');
103 foreach ($this->db_object->get_tz_polygons($ids) as $entity) {
104 if (self::_wn_PackedPoly($longitude, $latitude, $entity['polygon'])) {
105 return $entity['tzname'];
106 }
107 unset($entity);
108 }
109 return '';
110 }
111
112 /***********************************************************************************************************************/
113 /**
114 The winding-number algorithm courtesy of San Zhujun (https://gist.github.com/zhujunsan/81d6a2f05d590f618a5ad36f25666fc2),
115 operating on the existing packed native-double format with the same edge/vertex comparisons.
116 Decode 1,024 points at a time: per-point unpack calls are expensive, while unpacking an entire large polygon
117 creates a large PHP hash table. Carry the preceding point across blocks, including the closing edge.
118 */
119 private static function _wn_PackedPoly($longitude, $latitude, $polygon) {
120 $bytes = strlen($polygon);
121 if ($bytes < 32) {
122 return false;
123 }
124 $last = unpack('dx/dy', $polygon, $bytes - 16);
125 $previousX = $last['x'];
126 $previousY = $last['y'];
127 $winding = 0;
128 for ($offset = 0; $offset < $bytes; $offset += 16384) {
129 $count = min(2048, intdiv($bytes - $offset, 8));
130 $coordinates = unpack('d'.$count, $polygon, $offset);
131 for ($i = 1; $i < $count; $i += 2) {
132 $x = $coordinates[$i];
133 $y = $coordinates[$i + 1];
134 if ($previousX <= $longitude) {
135 if ($x > $longitude &&
136 ($y - $previousY) * ($longitude - $previousX) - ($latitude - $previousY) * ($x - $previousX) > 0) {
137 ++$winding;
138 }
139 } elseif ($x <= $longitude &&
140 ($y - $previousY) * ($longitude - $previousX) - ($latitude - $previousY) * ($x - $previousX) < 0) {
141 --$winding;
142 }
143 $previousX = $x;
144 $previousY = $y;
145 }
146 }
147 return 0 != $winding;
148 }
