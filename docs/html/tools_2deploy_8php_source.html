40function deployIdentifier(string $name, string $driver): string {
41 $quote = $driver === 'pgsql' ? '"' : '`';
42 return $quote.str_replace($quote, $quote.$quote, $name).$quote;

56/** \brief This opens the configured application or administrative database. \throws PDOException on failure. */
57function deployConnection(array $settings, bool $administrative = false): PDO {
58 $driver = $settings['driver'];
59 $dsn = $driver.':host='.$settings['host'].';port='.$settings['port'];
60 if (!$administrative || $driver === 'pgsql') {
61 $dsn .= ';dbname='.($administrative ? $settings['admin_database'] : $settings['database']);
62 }
63 $dsn .= $driver === 'mysql' ? ';charset=utf8mb4' : ';options=--client_encoding=UTF8';
64 return new PDO($dsn, $settings['user'], $settings['password'], [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]);
65}

68/** \brief This prevents two installers from loading the same database concurrently. */
69function deployLock(PDO $connection): void {
70 $sql = $connection->getAttribute(PDO::ATTR_DRIVER_NAME) === 'pgsql'
71 ? "SELECT pg_try_advisory_lock(hashtext('LGV_TZ_Lookup.install'))"
72 : "SELECT GET_LOCK(CONCAT('lgv_tz_', SHA2(DATABASE(), 224)), 0)";
73 if (!$connection->query($sql)->fetchColumn()) {
74 throw new RuntimeException('Another timezone installation is using this database.');

79/** \brief This checks for an existing table without attempting to create or replace it. */
80function deployTableExists(PDO $connection, string $table): bool {
81 $schema = $connection->getAttribute(PDO::ATTR_DRIVER_NAME) === 'pgsql' ? 'current_schema()' : 'DATABASE()';
82 $statement = $connection->prepare('SELECT COUNT(*) FROM information_schema.tables WHERE table_schema = '.$schema.' AND table_name = ?');
83 $statement->execute([$table]);
84 return (int)$statement->fetchColumn() > 0;
85}

88/**
89 \brief This asks for a setting, with a default and optional password hiding.
90 \throws RuntimeException if terminal input ends, or password echo cannot be disabled.
91*/
92function deployAsk($terminal, string $label, string $default = '', bool $hidden = false): string {
93 fwrite(STDOUT, $label.($default === '' ? '' : ' ['.$default.']').': ');
94 $mode = null;
95 if ($hidden && stream_isatty($terminal)) {
96 $process = proc_open(['stty', '-g'], [0 => $terminal, 1 => ['pipe', 'w'], 2 => STDERR], $pipes);
97 if (false === $process) { throw new RuntimeException('Could not read terminal settings.'); }
98 $mode = trim(stream_get_contents($pipes[1]));
99 fclose($pipes[1]);
100 if (proc_close($process) !== 0 || $mode === '') { throw new RuntimeException('Could not read terminal settings.'); }
101 $process = proc_open(['stty', '-echo'], [0 => $terminal, 1 => STDOUT, 2 => STDERR], $pipes);
102 if (false === $process || proc_close($process) !== 0) { throw new RuntimeException('Could not hide password input.'); }
103 }
104 try {
105 if (stream_isatty($terminal)) {
106 // fgets() can retry an interrupted terminal read indefinitely. Poll readiness so Ctrl+C can unwind
107 // through the finally block and restore password echo before the shell starts cleanup.
108 do {
109 $read = [$terminal]; $write = $except = null;
110 $ready = @stream_select($read, $write, $except, 0, 250000);
111 } while ($ready === false || $ready === 0);
112 }
113 $answer = fgets($terminal);
114 if (false === $answer) { throw new RuntimeException('Installation input ended before setup was complete.'); }
115 $answer = $hidden ? rtrim($answer, "\r\n") : trim($answer);
116 return $answer === '' ? $default : $answer;
117 } finally {
118 if ($mode !== null) {
119 $process = proc_open(['stty', $mode], [0 => $terminal, 1 => STDOUT, 2 => STDERR], $pipes);
120 if (is_resource($process)) { proc_close($process); }
121 fwrite(STDOUT, "\n");
122 }
123 }
124}

127/** \brief Build the browser URL from the selected web directory's URL and the new service subdirectory. */
128function deployServiceUrl(string $webUrl, string $endpoint): string {
129 $webUrl = trim($webUrl);
130 $parts = parse_url($webUrl);
131 if (filter_var($webUrl, FILTER_VALIDATE_URL) === false || !is_array($parts) ||
132 !in_array(strtolower($parts['scheme'] ?? ''), ['http', 'https'], true) ||
133 isset($parts['user']) || isset($parts['pass']) || isset($parts['query']) || isset($parts['fragment']) ||
134 str_contains($webUrl, '\\')) {
135 throw new InvalidArgumentException('Enter the full http:// or https:// URL of the web directory, without credentials, a query, or a fragment.');
136 }
137 if (!preg_match('/^[a-zA-Z0-9_-]+$/D', $endpoint)) {
138 throw new InvalidArgumentException('Use a simple service directory name.');
139 }
140 return rtrim($webUrl, '/').'/'.$endpoint.'/';
141}

144/** \brief This removes a directory tree without following symlinks. */
145function deployRemoveTree(string $directory): void {
146 if (!is_dir($directory) || is_link($directory)) { return; }
147 $files = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($directory, FilesystemIterator::SKIP_DOTS), RecursiveIteratorIterator::CHILD_FIRST);
148 foreach ($files as $file) {
149 $ok = $file->isDir() && !$file->isLink() ? rmdir($file->getPathname()) : unlink($file->getPathname());
150 if (!$ok) { throw new RuntimeException('Could not remove '.$file->getPathname()); }
151 }
152 if (!rmdir($directory)) { throw new RuntimeException('Could not remove '.$directory); }
153}

201/** \brief Copy an installation for staging, preserving additional files, permissions, and symbolic links. */
202function deployCopyTree(string $source, string $destination): void {
203 $mode = fileperms($source) & 0777;
204 if (!mkdir($destination, $mode) || !chmod($destination, $mode)) { throw new RuntimeException('Could not stage the installation.'); }
205 foreach (new DirectoryIterator($source) as $entry) {
206 if ($entry->isDot()) { continue; }
207 $target = $destination.'/'.$entry->getFilename();
208 if ($entry->isLink()) { $ok = symlink(readlink($entry->getPathname()), $target); }
209 elseif ($entry->isDir()) { deployCopyTree($entry->getPathname(), $target); continue; }
210 else { $ok = copy($entry->getPathname(), $target) && chmod($target, $entry->getPerms() & 0777); }
211 if (!$ok) { throw new RuntimeException('Could not stage '.$entry->getFilename()); }
212 }

226/** \brief Restore prior directories after an interrupted refresh, or remove backups after publication. Never changes database tables. */
227function deployRefreshCleanup(string $work, array $state): void {
228 $verify = static function(string $path, string $token): void {
229 if (is_link($path) || !is_dir($path) || !is_file($path.'/.lgv-tz-install-owner') || is_link($path.'/.lgv-tz-install-owner') ||
230 !hash_equals($token, trim(file_get_contents($path.'/.lgv-tz-install-owner')))) {
231 throw new RuntimeException('Refresh directory ownership differs; refusing recovery for '.$path);
232 }
233 };
234 foreach (['private', 'public'] as $key) {
235 $backup = $state[$key.'_backup'];
236 if (!file_exists($backup) && !is_link($backup)) { continue; }
237 $verify($backup, $state['previous_token']);
238 if (empty($state['published'])) {
239 if (file_exists($state[$key]) || is_link($state[$key])) {
240 $verify($state[$key], $state['token']);
241 deployRemoveTree($state[$key]);
242 }
243 if (!rename($backup, $state[$key])) { throw new RuntimeException('Could not restore the prior installation.'); }
244 } else {
245 $verify($state[$key], $state['token']);
246 deployRemoveTree($backup);
247 }
248 }
249 foreach (['private_stage', 'public_stage'] as $key) {
250 if (!file_exists($state[$key]) && !is_link($state[$key])) { continue; }
251 $verify($state[$key], $state['token']);
252 deployRemoveTree($state[$key]);
253 }
254 unlink($work.'/deployment.json');
255}

258/** \brief Refresh owned installed code using existing polygons, retaining the configuration and secret. */
259function deployRefresh(string $work, array $settings, string $private, string $public, int $group,
260 string $serviceUrl, array $existing, callable $install): void {
261 $connection = deployConnection($settings);
262 deployLock($connection);
263 if (deployTableExists($connection, 'lgv_tz_install_guard')) {
264 throw new RuntimeException('An installation recovery table remains; finish its cleanup before refreshing.');
265 }
266 $count = (int)$connection->query('SELECT COUNT(*) FROM timezones')->fetchColumn();
267 if ($count === 0) { throw new RuntimeException('The existing boundary table is empty; refusing to refresh.'); }
268 $suffix = bin2hex(random_bytes(12));
269 $state = ['refresh' => true, 'token' => bin2hex(random_bytes(32)), 'previous_token' => $existing['token'],
270 'private' => $private, 'public' => $public, 'private_stage' => dirname($private).'/.lgv-tz-install-'.$suffix,
271 'public_stage' => dirname($public).'/.lgv-tz-install-'.$suffix,
272 'private_backup' => dirname($private).'/.lgv-tz-previous-'.$suffix,
273 'public_backup' => dirname($public).'/.lgv-tz-previous-'.$suffix, 'published' => false];
274 deploySave($work, $state);
275 foreach (['private', 'public'] as $key) {
276 // Establish ownership before copying, so cleanup can always identify even an incomplete stage.
277 if (!mkdir($state[$key.'_stage'], 0700) ||
278 file_put_contents($state[$key.'_stage'].'/.lgv-tz-install-owner', $state['token']) === false) {
279 throw new RuntimeException('Could not create refresh staging directories.');
280 }
281 foreach (new DirectoryIterator($state[$key]) as $entry) {
282 if ($entry->isDot() || $entry->getFilename() === '.lgv-tz-install-owner') { continue; }
283 $target = $state[$key.'_stage'].'/'.$entry->getFilename();
284 if ($entry->isLink()) { $ok = symlink(readlink($entry->getPathname()), $target); }
285 elseif ($entry->isDir()) { deployCopyTree($entry->getPathname(), $target); continue; }
286 else { $ok = copy($entry->getPathname(), $target) && chmod($target, $entry->getPerms() & 0777); }
287 if (!$ok) { throw new RuntimeException('Could not stage the existing installation.'); }
288 }
289 }
290 $privateStage = $state['private_stage'];
291 $publicStage = $state['public_stage'];
292 echo "Refreshing installed code; keeping the existing boundaries, configuration, and secret.\n";
293 deployRemoveTree($privateStage.'/app');
294 mkdir($privateStage.'/app', 0755);
295 $install($privateStage.'/app', $settings['driver'], dirname(__DIR__));
296 $package = Composer\InstalledVersions::getInstallPath(LGV_TZ_Lookup_Setup::PACKAGE);
297 $database = new LGV_TZ_Lookup_Database($settings['database'], $settings['user'], $settings['password'],
298 $settings['driver'], $settings['host'], $settings['port']);
299 require $package.'/src/TestLocations.php';
300 $lookup = new LGV_TZ_Lookup_Query($database);
301 $failures = 0;
302 foreach ($test_locations_param_array as $case) {
303 $lng = $case['params']['lng']; $lat = $case['params']['lat'];
304 while ($lng < -180) { $lng += 360; } while ($lng > 180) { $lng -= 360; }
305 while ($lat < -90) { $lat += 180; } while ($lat > 90) { $lat -= 180; }
306 if ($lookup->get_tz($lng, $lat) !== $case['result']) { ++$failures; }
307 }
308 if ($failures) { throw new RuntimeException($failures.' known-location tests failed; the existing installation was retained.'); }
309 foreach (['update.php', 'Setup.php'] as $file) {
310 if (!copy(__DIR__.'/'.$file, $privateStage.'/tools/'.$file)) { throw new RuntimeException('Could not refresh updater code.'); }
311 }
312 if (!copy(dirname(__DIR__).'/update.sh', $privateStage.'/update.sh')) { throw new RuntimeException('Could not refresh the updater command.'); }
313 $source = file_get_contents(__FILE__);
314 $license = explode("/***************************************************************************************************************************/\n/**\n \\file", $source, 2)[0];
315 $relativePackage = substr($package, strlen($privateStage));
316 if (file_put_contents($publicStage.'/index.php', deployEndpointCode($private, $relativePackage, $license)) === false) {
317 throw new RuntimeException('Could not write the refreshed public entry point.');
318 }
319 foreach (['composer.phar', 'composer.sha256'] as $file) { if (is_file($privateStage.'/app/'.$file)) { unlink($privateStage.'/app/'.$file); } }
320 foreach (['composer-cache', 'composer-home'] as $dir) { if (is_dir($privateStage.'/app/'.$dir)) { deployRemoveTree($privateStage.'/app/'.$dir); } }
321 $files = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($privateStage.'/app', FilesystemIterator::SKIP_DOTS), RecursiveIteratorIterator::SELF_FIRST);
322 foreach ($files as $file) {
323 if (!$file->isLink() && !chmod($file->getPathname(), $file->isDir() || ($file->getPerms() & 0111) ? 0755 : 0644)) {
324 throw new RuntimeException('Could not set refreshed code permissions.');
325 }
326 }
327 if (!chgrp($privateStage, $group) || !chgrp($privateStage.'/config.php', $group) ||
328 !chmod($privateStage, 0750) || !chmod($privateStage.'/config.php', 0640) ||
329 !chmod($privateStage.'/update.sh', 0755) || !chmod($publicStage, 0755) || !chmod($publicStage.'/index.php', 0644)) {
330 throw new RuntimeException('Could not set refreshed installation permissions.');
331 }
332 $metadata = $existing['metadata'];
333 $metadata['service_url'] = $serviceUrl;
334 $metadata['known_location_checks'] = count($test_locations_param_array);
335 if (file_put_contents($privateStage.'/installation.json', json_encode($metadata, JSON_PRETTY_PRINT | JSON_THROW_ON_ERROR)) === false) {
336 throw new RuntimeException('Could not write refreshed installation metadata.');
337 }
338 foreach (['private', 'public'] as $key) {
339 if (!rename($state[$key], $state[$key.'_backup']) || !rename($state[$key.'_stage'], $state[$key])) {
340 throw new RuntimeException('Could not publish refreshed code; recovery will restore the prior installation.');
341 }
342 }
343 $state['published'] = true;
344 deploySave($work, $state);
345 deployRefreshCleanup($work, $state);
346 $secretQuery = $existing['secret'] === '' ? '' : '&secret='.rawurlencode($existing['secret']);
347 echo 'Refreshed installed code; retained '.$count.' polygons; '.count($test_locations_param_array)." known locations passed.\n";
348 echo 'Public endpoint: '.$public.'/index.php'."\nPrivate configuration: ".$private.'/config.php'."\n";
349 echo 'Service URL: '.$serviceUrl."\n";
350 echo 'Example request: '.$serviceUrl.'?ll=-77.036543,38.895037'.$secretQuery."\n";
351 echo 'Test request: '.$serviceUrl.'?test'.$secretQuery."\n";
352}

355/**
356 \brief This removes installer downloads, and rolls back only installation-owned resources on failure.
357 A completed deployment retains its database and files. The guard token authorizes removal of an incomplete
358 timezones table; an unrelated table or installation is never adopted. Failed cleanup retains recovery state.
359*/
360function deployCleanup(string $work): void {
361 foreach (['boundaries.zip', 'boundaries.json'] as $file) {
362 if (is_file($work.'/'.$file) && !unlink($work.'/'.$file)) { throw new RuntimeException('Could not remove downloaded '.$file); }
363 }
364 $file = $work.'/deployment.json';
365 if (!is_file($file)) { return; }
366 $state = json_decode(file_get_contents($file), true, 512, JSON_THROW_ON_ERROR);
367 if (!empty($state['refresh'])) { deployRefreshCleanup($work, $state); return; }
368 if (!empty($state['guard_created'])) {
369 $connection = deployConnection($state['settings']);
370 deployLock($connection);
371 if (deployTableExists($connection, 'lgv_tz_install_guard')) {
372 $statement = $connection->prepare('SELECT token FROM lgv_tz_install_guard');
373 $statement->execute();
374 $token = $statement->fetchColumn();
375 $statement->closeCursor();
376 $statement = null;
377 if (!is_string($token) || !hash_equals($state['token'], $token)) {
378 throw new RuntimeException('Installation ownership differs; refusing to remove database tables.');
379 }
380 if (empty($state['published'])) { $connection->exec('DROP TABLE IF EXISTS timezones'); }
381 $connection->exec('DROP TABLE lgv_tz_install_guard');
382 } elseif (empty($state['published'])) {
383 throw new RuntimeException('Installation ownership record is missing; refusing to remove database tables.');
384 }
385 $state['guard_created'] = false;
386 deploySave($work, $state);
387 $connection = null;
388 }
389 if (empty($state['published']) && !empty($state['database_created'])) {
390 deployConnection($state['settings'], true)->exec('DROP DATABASE '.deployIdentifier($state['settings']['database'], $state['settings']['driver']));
391 $state['database_created'] = false;
392 deploySave($work, $state);
393 }
394 $paths = [$state['private_stage'], $state['public_stage']];
395 if (empty($state['published'])) { $paths = array_merge($paths, [$state['private'], $state['public']]); }
396 foreach ($paths as $path) {
397 if (!is_dir($path)) { continue; }
398 $marker = $path.'/.lgv-tz-install-owner';
399 if (!is_file($marker) || !hash_equals($state['token'], trim(file_get_contents($marker)))) {
400 throw new RuntimeException('Directory ownership differs; refusing to remove '.$path);
401 }
402 deployRemoveTree($path);
403 }
404 unlink($file);
405}

408/**
409 \brief This gathers settings, loads and tests the Composer package, then publishes the endpoint.
410 The selected web directory must exist. Existing installer-owned directories can be refreshed without reloading boundaries;
411 private files must be outside the web document root. Existing timezones tables are refused before loading.
412 \throws Exception if setup, validation, or publication fails.
413*/
414function deployRun(string $work, bool $noSecret = false, $terminal = null, ?callable $install = null): void {
415 $started = hrtime(true);
416 foreach (['curl', 'zip', 'mbstring', 'ctype', 'pdo', 'posix'] as $extension) {
417 if (!extension_loaded($extension)) { throw new RuntimeException('The PHP '.$extension.' extension is required.'); }
418 }
419 if (!function_exists('proc_open')) { throw new RuntimeException('PHP proc_open is required.'); }
420 $terminal = $terminal ?? @fopen('/dev/tty', 'r+');
421 if (!is_resource($terminal)) { throw new RuntimeException('Run the installer from an interactive terminal.'); }
422 echo "LGV_TZ_Lookup server installation\n\n";
423 $driver = strtolower(deployAsk($terminal, 'Database driver (mysql or pgsql)', 'mysql'));
424 if (!in_array($driver, ['mysql', 'pgsql'], true) || !extension_loaded('pdo_'.$driver)) {
425 throw new RuntimeException('Choose mysql or pgsql, with its PHP PDO extension enabled.');
426 }
427 $settings = ['driver' => $driver, 'host' => deployAsk($terminal, 'Database host', 'localhost')];
428 $port = deployAsk($terminal, 'Database port', $driver === 'pgsql' ? '5432' : '3306');
429 if (!ctype_digit($port) || (int)$port < 1 || (int)$port > 65535) { throw new RuntimeException('Port must be between 1 and 65535.'); }
430 $settings['port'] = (int)$port;
431 $settings['database'] = deployAsk($terminal, 'Database name', 'tz_lookup');
432 $maxName = $driver === 'pgsql' ? 63 : 64;
433 if (!preg_match('/^[a-zA-Z0-9_-]{1,'.$maxName.'}$/D', $settings['database'])) {
434 throw new RuntimeException('Use a database name containing letters, numbers, underscores, or hyphens.');
435 }
436 $settings['user'] = deployAsk($terminal, 'Database user', $driver === 'pgsql' ? (getenv('USER') ?: get_current_user()) : 'root');
437 $settings['password'] = deployAsk($terminal, 'Database password (input hidden; empty is allowed)', '', true);
438 $settings['admin_database'] = $driver === 'pgsql' ? deployAsk($terminal, 'Administrative database (used if creating a database)', 'postgres') : '';
439 $create = strtolower(deployAsk($terminal, 'Create the database if it does not exist? (y/n)', 'y')) === 'y';
