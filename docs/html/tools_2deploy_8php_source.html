27 \brief This installs a tested HTTP timezone service, with private configuration and a default server secret.
28
29 deploy.sh supplies a private working directory. The worker asks for settings through the terminal, installs
30 the Composer package, loads current boundaries, and checks known locations before publishing a small endpoint.
31 A separate cleanup process removes downloads and rolls back an incomplete installation after errors or OOM.
32 Database ownership is verified with a random guard token before cleanup can remove a newly loaded table.

40function deployIdentifier(string $name, string $driver): string {
41 $quote = $driver === 'pgsql' ? '"' : '`';
42 return $quote.str_replace($quote, $quote.$quote, $name).$quote;

56/** \brief This opens the configured application or administrative database. \throws PDOException on failure. */
57function deployConnection(array $settings, bool $administrative = false): PDO {
58 $driver = $settings['driver'];
59 $dsn = $driver.':host='.$settings['host'].';port='.$settings['port'];
60 if (!$administrative || $driver === 'pgsql') {
61 $dsn .= ';dbname='.($administrative ? $settings['admin_database'] : $settings['database']);
62 }
63 $dsn .= $driver === 'mysql' ? ';charset=utf8mb4' : ';options=--client_encoding=UTF8';
64 return new PDO($dsn, $settings['user'], $settings['password'], [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]);
65}

68/** \brief This prevents two installers from loading the same database concurrently. */
69function deployLock(PDO $connection): void {
70 $sql = $connection->getAttribute(PDO::ATTR_DRIVER_NAME) === 'pgsql'
71 ? "SELECT pg_try_advisory_lock(hashtext('LGV_TZ_Lookup.install'))"
72 : "SELECT GET_LOCK(CONCAT('lgv_tz_', SHA2(DATABASE(), 224)), 0)";
73 if (!$connection->query($sql)->fetchColumn()) {
74 throw new RuntimeException('Another timezone installation is using this database.');
75 }

79/** \brief This checks for an existing table without attempting to create or replace it. */
80function deployTableExists(PDO $connection, string $table): bool {
81 $schema = $connection->getAttribute(PDO::ATTR_DRIVER_NAME) === 'pgsql' ? 'current_schema()' : 'DATABASE()';
82 $statement = $connection->prepare('SELECT COUNT(*) FROM information_schema.tables WHERE table_schema = '.$schema.' AND table_name = ?');
83 $statement->execute([$table]);
84 return (int)$statement->fetchColumn() > 0;

88/**
89 \brief This asks for a setting, with a default and optional password hiding.
90 \throws RuntimeException if terminal input ends, or password echo cannot be disabled.
91*/
92function deployAsk($terminal, string $label, string $default = '', bool $hidden = false): string {
93 fwrite(STDOUT, $label.($default === '' ? '' : ' ['.$default.']').': ');
94 $mode = null;
95 if ($hidden && stream_isatty($terminal)) {
96 $process = proc_open(['stty', '-g'], [0 => $terminal, 1 => ['pipe', 'w'], 2 => STDERR], $pipes);
97 if (false === $process) { throw new RuntimeException('Could not read terminal settings.'); }
98 $mode = trim(stream_get_contents($pipes[1]));
99 fclose($pipes[1]);
100 if (proc_close($process) !== 0 || $mode === '') { throw new RuntimeException('Could not read terminal settings.'); }
101 $process = proc_open(['stty', '-echo'], [0 => $terminal, 1 => STDOUT, 2 => STDERR], $pipes);
102 if (false === $process || proc_close($process) !== 0) { throw new RuntimeException('Could not hide password input.'); }
103 }
104 try {
105 if (stream_isatty($terminal)) {
106 // fgets() can retry an interrupted terminal read indefinitely. Poll readiness so Ctrl+C can unwind
107 // through the finally block and restore password echo before the shell starts cleanup.
108 do {
109 $read = [$terminal]; $write = $except = null;
110 $ready = @stream_select($read, $write, $except, 0, 250000);
111 } while ($ready === false || $ready === 0);
112 }
113 $answer = fgets($terminal);
114 if (false === $answer) { throw new RuntimeException('Installation input ended before setup was complete.'); }
115 $answer = $hidden ? rtrim($answer, "\r\n") : trim($answer);
116 return $answer === '' ? $default : $answer;
117 } finally {
118 if ($mode !== null) {
119 $process = proc_open(['stty', $mode], [0 => $terminal, 1 => STDOUT, 2 => STDERR], $pipes);
120 if (is_resource($process)) { proc_close($process); }
121 fwrite(STDOUT, "\n");
122 }
123 }
124}

127/** \brief This removes a directory tree without following symlinks. */
128function deployRemoveTree(string $directory): void {
129 if (!is_dir($directory) || is_link($directory)) { return; }
130 $files = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($directory, FilesystemIterator::SKIP_DOTS), RecursiveIteratorIterator::CHILD_FIRST);
131 foreach ($files as $file) {
132 $ok = $file->isDir() && !$file->isLink() ? rmdir($file->getPathname()) : unlink($file->getPathname());
133 if (!$ok) { throw new RuntimeException('Could not remove '.$file->getPathname()); }
134 }
135 if (!rmdir($directory)) { throw new RuntimeException('Could not remove '.$directory); }

139/**
140 \brief This removes installer downloads, and rolls back only installation-owned resources on failure.
141 A completed deployment retains its database and files. The guard token authorizes removal of an incomplete
142 timezones table; an unrelated table or installation is never adopted. Failed cleanup retains recovery state.

144function deployCleanup(string $work): void {
145 foreach (['boundaries.zip', 'boundaries.json'] as $file) {
146 if (is_file($work.'/'.$file) && !unlink($work.'/'.$file)) { throw new RuntimeException('Could not remove downloaded '.$file); }
147 }
148 $file = $work.'/deployment.json';
149 if (!is_file($file)) { return; }
150 $state = json_decode(file_get_contents($file), true, 512, JSON_THROW_ON_ERROR);
151 if (!empty($state['guard_created'])) {
152 $connection = deployConnection($state['settings']);
153 deployLock($connection);
154 if (deployTableExists($connection, 'lgv_tz_install_guard')) {
155 $statement = $connection->prepare('SELECT token FROM lgv_tz_install_guard');
156 $statement->execute();
157 $token = $statement->fetchColumn();
158 $statement->closeCursor();
159 $statement = null;
160 if (!is_string($token) || !hash_equals($state['token'], $token)) {
161 throw new RuntimeException('Installation ownership differs; refusing to remove database tables.');
162 }
163 if (empty($state['published'])) { $connection->exec('DROP TABLE IF EXISTS timezones'); }
164 $connection->exec('DROP TABLE lgv_tz_install_guard');
165 } elseif (empty($state['published'])) {
166 throw new RuntimeException('Installation ownership record is missing; refusing to remove database tables.');
167 }
168 $state['guard_created'] = false;
169 deploySave($work, $state);
170 $connection = null;
171 }
172 if (empty($state['published']) && !empty($state['database_created'])) {
173 deployConnection($state['settings'], true)->exec('DROP DATABASE '.deployIdentifier($state['settings']['database'], $state['settings']['driver']));
174 $state['database_created'] = false;
175 deploySave($work, $state);
176 }
177 $paths = [$state['private_stage'], $state['public_stage']];
178 if (empty($state['published'])) { $paths = array_merge($paths, [$state['private'], $state['public']]); }
179 foreach ($paths as $path) {
180 if (!is_dir($path)) { continue; }
181 $marker = $path.'/.lgv-tz-install-owner';
182 if (!is_file($marker) || !hash_equals($state['token'], trim(file_get_contents($marker)))) {
183 throw new RuntimeException('Directory ownership differs; refusing to remove '.$path);
184 }
185 deployRemoveTree($path);
186 }
187 unlink($file);
188}
