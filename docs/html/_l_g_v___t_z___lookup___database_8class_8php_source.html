108 /**
109 This is a sort callback that sorts, so the smaller domain gets put first.
110
111 \returns: True, if $in_A < $inB
112 */
113 private static function _sort_by_domainSize($in_A, ///< The first entity
114 $in_B ///< The second entity
115 ) {
116 $ret = false;
117
118 if (isset($in_A["east"]) &&
119 isset($in_A["west"]) &&
120 isset($in_A["north"]) &&
121 isset($in_A["south"]) &&
122 isset($in_B["east"]) &&
123 isset($in_B["west"]) &&
124 isset($in_B["north"]) &&
125 isset($in_B["south"])) {
126 $areaA = abs(floatval($in_A["east"] - $in_A["west"])) * abs(floatval($in_A["south"] - $in_A["north"]));
127 $areaB = abs(floatval($in_B["east"] - $in_B["west"])) * abs(floatval($in_B["south"] - $in_B["north"]));
128 return ($areaA < $areaB) ? -1 : (($areaA < $areaB) ? 1 : 0);
129 }
130
131 return $ret;
132 }

135 /**
136 This is a simple "brute force" array flattener. It rounds the values to six significant digits, as well.
137 */
138 static function _array_flatten( $inArray ///< The multi-dimensional array to be squished.
139 ) {
140 $resulting_array = array();
141 array_walk_recursive($inArray, function($inValue) use (&$resulting_array) { array_push($resulting_array, round($inValue, 6)); });
142 return $resulting_array;
143 }

146 /**
147 This uses our PDO instance to save the entity as a table row.
148 */
149 public function store_entity( $inEntity ///< The entity to be saved into the database.
150 ) {
151 $polygon_nested_array = array($inEntity->polygon);
152 $polygon_array = self::_array_flatten($polygon_nested_array);
153 $polygon = implode(array_map(function($inValue) { return pack('d*', $inValue); }, $polygon_array));
154 $sql = 'INSERT INTO '.$this->_load_table.' (tzname, east, west, north, south, polygon) VALUES (?, ?, ?, ?, ?, ?)';
155 $params = [$inEntity->tzID, $inEntity->domainRect['east'], $inEntity->domainRect['west'], $inEntity->domainRect['north'], $inEntity->domainRect['south'], $polygon];
156 // Packed doubles contain zero bytes; PostgreSQL must receive them as binary, not text parameters.
157 $this->pdo_instance->preparedStatement($sql, $params, false, [5 => PDO::PARAM_LOB]);
158 }

161 /**
162 This queries the database, and returns the TZ IDs.
163
164 \returns: An array of integers, which are the database IDs of table rows that have the long/lat in their domainRects.
165 */
166 public function get_tz_ids( $in_lng, ///< The longitude
167 $in_lat ///< The latitude
168 ) {
169 $sql = "SELECT id, tzname, east, west, north, south FROM timezones WHERE east>=? AND west<=? AND north>=? AND south<=?";
170
171 $params = [$in_lng, $in_lng, $in_lat, $in_lat];
172 $ret = $this->pdo_instance->preparedStatement($sql, $params, true);
173 if (1 < count($ret)) {
174 usort($ret, 'LGV_TZ_Lookup_Database::_sort_by_domainSize');
175 }
176 return $ret;
177 }

180 /**
181 \returns: An array of simple arrays ([tzname, polygon]), which are the database entities of table rows that correspond to the IDs passed in.
182 */
183 public function get_tz_entities( $in_id_list ///< A list of IDs for the rows we want to check.
184 ) {
185 $sql = "SELECT tzname, polygon FROM timezones WHERE";
186 $idSQL = "";
187 $params = [];
188
189 foreach ($in_id_list as $id) {
190 if (!empty($idSQL)) {
191 $idSQL .= " OR";
192 }
193
194 $idSQL .= " id=?";
195 $params[] = $id;
196 }
197
198 $entities = [];
199
200 if (!empty($params)) {
201 $ret = $this->pdo_instance->preparedStatement($sql.$idSQL, $params, true);
202
203 if (!empty($ret)) {
204 foreach($ret as $row) {
205 $entity = ['tzname' => $row['tzname'], 'polygon' => unpack('d*', $row['polygon'])];
206 $entity['polygon'] = array_chunk(array_values($entity['polygon']), 2);
207 $entities[] = $entity;
208 }
209 }
210 }
211
212 return $entities;
213 }

216 /**
217 Yield the existing binary polygons without unpacking them. Primary-key order makes first-match precedence
218 explicit, matching the existing MySQL primary-key candidate scan.
219 */
220 public function get_tz_polygons($in_id_list) {
221 if (empty($in_id_list)) {
222 return;
223 }
224 $placeholders = implode(',', array_fill(0, count($in_id_list), '?'));
225 $sql = 'SELECT tzname, polygon FROM timezones WHERE id IN ('.$placeholders.') ORDER BY id';
226 yield from $this->pdo_instance->preparedRows($sql, array_values($in_id_list), false);
227 }
