25/**
26 \brief This class provides a genericized interface to the [PHP PDO](http://us.php.net/pdo) toolkit.
27
28 This is a PDO abstraction class, derived from the Badger Hardened Baseline Database Component
29
30 This supports MySQL and PostgreSQL, including binary parameters and bounded polygon reads.
31 */
32class LGV_TZ_Lookup_PDO {
33 /// \brief Internal PDO object
34 private $_pdo = NULL;
35 /// \brief The type of PDO driver we are configured for.
36 var $driver_type = NULL;
37 /// \brief This holds the integer ID of the last AUTO_INCREMENT insert.
38 var $last_insert = NULL;
39
40 /***********************************************************************************************************************/
41 /***********************/
42 /**
43 \brief Initializes connection param class members.
44
45 Must be called BEFORE any attempts to connect to or query a database. This uses UTF8, as the charset.
46
47 Will destroy previous connection (if one exists).
48 */
49 public function __construct( $inDatabase, ///< database name
50 $inUser, ///< database user
51 $inPassword, ///< database password
52 $inDriver, ///< database server type
53 $inHost, ///< database server host
54 $inPort ///< database TCP port
55 ) {
56 $this->_pdo = NULL;
57 $this->driver_type = strtolower($inDriver);
58
59 if (!in_array($this->driver_type, ['mysql', 'pgsql'], true)) {
60 throw new InvalidArgumentException('Supported database drivers are mysql and pgsql.');
61 }
62
63 $inPort = $inPort ?? ('pgsql' === $this->driver_type ? 5432 : 3306);
64 $dsn = $this->driver_type . ':host=' . $inHost . ';dbname=' . $inDatabase . ';port=' . strval($inPort);
65 $dsn .= 'mysql' === $this->driver_type ? ';charset=utf8' : ';options=--client_encoding=UTF8';
66
67 try {
68 $this->_pdo = new PDO($dsn, $inUser, $inPassword);
69 $this->_pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
70 $this->_pdo->setAttribute(PDO::ATTR_CASE, PDO::CASE_LOWER);
71 $this->_pdo->setAttribute(PDO::ATTR_EMULATE_PREPARES, 'mysql' === $this->driver_type);
72 if ('pgsql' === $this->driver_type) {
73 // Execute once without a separate PREPARE round trip; retain native binary parameter binding.
74 $attribute = class_exists('Pdo\\Pgsql') ? \Pdo\Pgsql::ATTR_DISABLE_PREPARES : PDO::PGSQL_ATTR_DISABLE_PREPARES;
75 $this->_pdo->setAttribute($attribute, true);
76 }
77 } catch (PDOException $exception) {
78 throw new Exception(__METHOD__ . '() ' . $exception->getMessage());
79 }
80 }
81
82 /***********************/
83 /**
84 \brief Wrapper for preparing and executing a PDOStatement
85
86 \throws Exception thrown if internal PDO exception is thrown
87 \returns true if execution is successful (and fetchResponse is false), or an array of associative arrays of results, if fetchResponse is true.
88 */
89 public function preparedStatement( $sql, ///< SQL statement to send (with question mark placeholders).
90 $params = array(), ///< Data for the placeholders. Default is an empty array.
91 $fetchResponse = false, ///< If true (default is false), then a fetch will be done, and a response returned.
92 $paramTypes = array() ///< Optional zero-based PDO parameter types for positional write parameters.
93 ) {
94 // A read needs no BEGIN/COMMIT round trips. Retain the array-returning public API.
95 if ($fetchResponse) {
96 return iterator_to_array($this->preparedRows($sql, $params), false);
97 }
98 if ( NULL == $this->_pdo ) {
99 throw new Exception(__METHOD__ . '()::' . __LINE__ . "\nNo PDO object!");
100 }
101
102 // Non-MySQL servers aren't fans of the backticks.
103 if ( 'mysql' != $this->driver_type ) {
104 $sql = str_ireplace('`', '', $sql);
105 }
106
107 $ownsTransaction = !$this->_pdo->inTransaction();
108 try {
109 if ($ownsTransaction) {
110 $this->_pdo->beginTransaction();
111 }
112
113 $stmt = $this->_pdo->prepare($sql);
114
115 if ( false == $stmt || -1 == $stmt ) {
116 throw new Exception(__METHOD__ . '()::' . __LINE__ . "\n" . print_r($stmt->errorInfo(), true));
117 }
118
119 if (empty($paramTypes)) {
120 $stmt->execute($params);
121 } else {
122 foreach (array_values($params) as $index => $value) {
123 $stmt->bindValue($index + 1, $value, $paramTypes[$index] ?? PDO::PARAM_STR);
124 }
125 $stmt->execute();
126 }
127 $stmt->closeCursor();
128
129 if ($ownsTransaction && $this->_pdo->inTransaction()) {
130 $this->_pdo->commit();
131 }
132
133 return true;
134 } catch (PDOException $exception) {
135 $this->last_insert = NULL;
136 if ($ownsTransaction && $this->_pdo->inTransaction()) {
137 $this->_pdo->rollBack();
138 }
139 throw new Exception(__METHOD__ . '()::' . __LINE__ . "\n" . $exception->getMessage(), 0, $exception);
140 }
141
142 return false;
143 }
144
145 /***********************************************************************************************************************/
146 /**
147 Yield rows one at a time and release the cursor, including when a lookup returns before consuming every row.
148 Polygon reads can disable MySQL buffering to avoid retaining every candidate blob. An unbuffered reader
149 must be closed before another query; the finally block drains its cursor and restores the connection mode.
150 PostgreSQL uses single-row fetching on PHP 8.5+, or a server cursor on earlier PHP releases. BYTEA streams
151 are read and closed here, so callers receive the same binary strings on either database.
152 */
153 public function preparedRows($sql, $params = array(), $buffered = true) {
154 if (NULL == $this->_pdo) {
155 throw new Exception(__METHOD__.'(): No PDO object!');
156 }
157 if ('mysql' != $this->driver_type) {
158 $sql = str_ireplace('`', '', $sql);
159 }
160 $stmt = NULL;
161 $bufferAttribute = NULL;
162 $previousBuffering = NULL;
163 $pgsqlStreaming = false;
164 $executed = false;
165 try {
166 if ('mysql' == $this->driver_type && !$buffered) {
167 // PHP 8.5 deprecates the PDO alias; the fallback supports the project's earlier PHP versions.
168 $bufferAttribute = class_exists('Pdo\\Mysql') ? \Pdo\Mysql::ATTR_USE_BUFFERED_QUERY : PDO::MYSQL_ATTR_USE_BUFFERED_QUERY;
169 $previousBuffering = $this->_pdo->getAttribute($bufferAttribute);
170 $this->_pdo->setAttribute($bufferAttribute, false);
171 }
172 $options = [];
173 if ('pgsql' === $this->driver_type && !$buffered) {
174 $pgsqlStreaming = PHP_VERSION_ID >= 80500;
175 $options = $pgsqlStreaming ? [PDO::ATTR_PREFETCH => 0] : [PDO::ATTR_CURSOR => PDO::CURSOR_SCROLL];
176 }
177 $stmt = $this->_pdo->prepare($sql, $options);
178 $stmt->execute($params);
179 $executed = true;
180 while ($row = $stmt->fetch(PDO::FETCH_ASSOC)) {
181 if ('pgsql' === $this->driver_type) {
182 foreach ($row as $column => $value) {
183 if (is_resource($value)) {
184 try {
185 $row[$column] = stream_get_contents($value);
186 if (false === $row[$column]) {
187 throw new RuntimeException('Could not read PostgreSQL binary column '.$column.'.');
188 }
189 } finally {
190 fclose($value);
191 }
192 }
193 }
194 unset($value);
195 }
196 yield $row;
197 unset($row);
198 }
199 } catch (PDOException $exception) {
200 throw new Exception(__METHOD__.'() '.$exception->getMessage(), 0, $exception);
201 } finally {
202 try {
203 if (NULL !== $stmt && false !== $stmt) {
204 if ($pgsqlStreaming && $executed) {
205 // PDO_PGSQL closeCursor() does not drain lazy results. Destroying an unfinished statement
206 // cancels its query, which can abort a caller's transaction. Consume without decoding columns.
207 while ($stmt->fetch(PDO::FETCH_BOUND)) { }
208 }
209 $stmt->closeCursor();
210 }
211 } finally {
212 if (NULL !== $previousBuffering) {
213 $this->_pdo->setAttribute($bufferAttribute, $previousBuffering);
214 }
215 }
216 }
217 }
218};
