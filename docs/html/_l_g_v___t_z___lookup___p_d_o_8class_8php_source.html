25/**
26 \brief This class provides a genericized interface to the [PHP PDO](http://us.php.net/pdo) toolkit.
27
28 This is a PDO abstraction class, derived from the Badger Hardened Baseline Database Component
29
30 This defaults to a standard localhost MySQL server (can be other types of servers).
31 */
32class LGV_TZ_Lookup_PDO {
33 /// \brief Internal PDO object
34 private $_pdo = NULL;
35 /// \brief The type of PDO driver we are configured for.
36 var $driver_type = NULL;
37 /// \brief This holds the integer ID of the last AUTO_INCREMENT insert.
38 var $last_insert = NULL;
39
40 /***********************************************************************************************************************/
41 /***********************/
42 /**
43 \brief Initializes connection param class members.
44
45 Must be called BEFORE any attempts to connect to or query a database. This uses UTF8, as the charset.
46
47 Will destroy previous connection (if one exists).
48 */
49 public function __construct( $inDatabase, ///< database name
50 $inUser, ///< database user
51 $inPassword, ///< database password
52 $inDriver, ///< database server type
53 $inHost, ///< database server host
54 $inPort ///< database TCP port
55 ) {
56 $this->_pdo = NULL;
57 $this->driver_type = strtolower($inDriver);
58
59 $dsn = $inDriver . ':host=' . $inHost . ';dbname=' . $inDatabase . ';charset=utf8;port=' . strval($inPort);
60
61 try {
62 $this->_pdo = new PDO($dsn, $inUser, $inPassword);
63 $this->_pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
64 $this->_pdo->setAttribute(PDO::ATTR_CASE, PDO::CASE_LOWER);
65 $this->_pdo->setAttribute(PDO::ATTR_EMULATE_PREPARES, true);
66 } catch (PDOException $exception) {
67 throw new Exception(__METHOD__ . '() ' . $exception->getMessage());
68 }
69 }
70
71 /***********************/
72 /**
73 \brief Wrapper for preparing and executing a PDOStatement
74
75 \throws Exception thrown if internal PDO exception is thrown
76 \returns true if execution is successful (and fetchResponse is false), or an array of associative arrays of results, if fetchResponse is true.
77 */
78 public function preparedStatement( $sql, ///< SQL statement to send (with question mark placeholders).
79 $params = array(), ///< Data for the placeholders. Default is an empty array.
80 $fetchResponse = false ///< If true (default is false), then a fetch will be done, and a response returned.
81 ) {
82 // A read needs no BEGIN/COMMIT round trips. Retain the array-returning public API.
83 if ($fetchResponse) {
84 return iterator_to_array($this->preparedRows($sql, $params), false);
85 }
86 if ( NULL == $this->_pdo ) {
87 throw new Exception(__METHOD__ . '()::' . __LINE__ . "\nNo PDO object!");
88 }
89
90 // Non-MySQL servers aren't fans of the backticks.
91 if ( 'mysql' != $this->driver_type ) {
92 $sql = str_ireplace('`', '', $sql);
93 }
94
95 try {
96 if ( !$this->_pdo->inTransaction() ) {
97 $this->_pdo->beginTransaction();
98 }
99
100 $stmt = $this->_pdo->prepare($sql);
101
102 if ( false == $stmt || -1 == $stmt ) {
103 throw new Exception(__METHOD__ . '()::' . __LINE__ . "\n" . print_r($stmt->errorInfo(), true));
104 }
105
106 $stmt->execute($params);
107
108 if ( $this->_pdo->inTransaction() ) {
109 $this->_pdo->commit();
110 }
111
112 return true;
113 } catch (PDOException $exception) {
114 $this->last_insert = NULL;
115 $this->_pdo->rollback();
116 throw new Exception(__METHOD__ . '()::' . __LINE__ . "\n" . $exception->getMessage());
117 }
118
119 return false;
120 }
121
122 /***********************************************************************************************************************/
123 /**
124 Yield rows one at a time and release the cursor, including when a lookup returns before consuming every row.
125 Polygon reads can disable MySQL buffering to avoid retaining every candidate blob. An unbuffered reader
126 must be closed before another query; the finally block drains its cursor and restores the connection mode.
127 */
128 public function preparedRows($sql, $params = array(), $buffered = true) {
129 if (NULL == $this->_pdo) {
130 throw new Exception(__METHOD__.'(): No PDO object!');
131 }
132 if ('mysql' != $this->driver_type) {
133 $sql = str_ireplace('`', '', $sql);
134 }
135 $stmt = NULL;
136 $bufferAttribute = NULL;
137 $previousBuffering = NULL;
138 try {
139 if ('mysql' == $this->driver_type && !$buffered) {
140 // PHP 8.5 deprecates the PDO alias; the fallback supports the project's earlier PHP versions.
141 $bufferAttribute = class_exists('Pdo\\Mysql') ? \Pdo\Mysql::ATTR_USE_BUFFERED_QUERY : PDO::MYSQL_ATTR_USE_BUFFERED_QUERY;
142 $previousBuffering = $this->_pdo->getAttribute($bufferAttribute);
143 $this->_pdo->setAttribute($bufferAttribute, false);
144 }
145 $stmt = $this->_pdo->prepare($sql);
146 $stmt->execute($params);
147 while ($row = $stmt->fetch(PDO::FETCH_ASSOC)) {
148 yield $row;
149 unset($row);
150 }
151 } catch (PDOException $exception) {
152 throw new Exception(__METHOD__.'() '.$exception->getMessage(), 0, $exception);
153 } finally {
154 try {
155 if (NULL !== $stmt && false !== $stmt) {
156 $stmt->closeCursor();
157 }
158 } finally {
159 if (NULL !== $previousBuffering) {
160 $this->_pdo->setAttribute($bufferAttribute, $previousBuffering);
161 }
162 }
163 }
164 }
165};
