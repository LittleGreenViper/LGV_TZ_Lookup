18function referenceContains(array $point, array $polygon): bool {
19 $winding = 0;
20 $count = count($polygon);
21 for ($i = 0; $i < $count; ++$i) {
22 $a = $polygon[$i];
23 $b = $polygon[($i + 1) % $count];
24 $side = ($b[1] - $a[1]) * ($point[0] - $a[0]) - ($point[1] - $a[1]) * ($b[0] - $a[0]);
25 if ($a[0] <= $point[0]) {
26 if ($b[0] > $point[0] && $side > 0) { ++$winding; }
27 } elseif ($b[0] <= $point[0] && $side < 0) {
28 --$winding;
29 }
30 }
31 return $winding !== 0;
32}
