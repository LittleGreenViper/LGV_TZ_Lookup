26 \file
27 \brief This is the PHP worker for the command-line Composer demo.
28
29 Call demo/run.sh to run the demo. The shell creates a temporary working directory, and calls this file in either
30 "run" or "cleanup" mode. In "run" mode, we install the released package through Composer, load the latest boundary
31 data, check the resulting database, run location tests, and display metrics. The lookup classes come from the
32 temporary Composer application, so this is a test of the installed package.
33
34 The separate "cleanup" invocation lets the shell remove the database and downloads, even if the loader's PHP

84/**
85 \brief This formats the timing statistics for one group of lookup calls.
86
87 We report the arithmetic mean, the median, the nearest-rank 95th percentile, and the slowest call.
88
89 \returns: A printable string. An empty group reports that no lookup measurements are available.
90*/
91function demoLatency(array $values ///< Individual lookup durations, in milliseconds.
92 ): string {
93 sort($values);
94 $count = count($values);
95 if ($count === 0) { return 'No lookups completed'; }
96 $median = ($values[(int)floor(($count - 1) / 2)] + $values[(int)floor($count / 2)]) / 2;
97 $p95 = $values[max(0, (int)ceil($count * 0.95) - 1)];
98 return sprintf('mean %.3f ms; median %.3f ms; p95 %.3f ms; max %.3f ms', array_sum($values) / $count, $median, $p95, max($values));
99}

102/**
103 \brief This prints the collected durations, data sizes, memory measurements, and test coverage.
104
105 Lookup timings cover calls to the package API, using an existing connection. They exclude downloading, loading,
106 and reference-geometry calculations. PHP memory measurements describe this worker, and do not include the database server
107 or the separate Composer process. Large-polygon audit counters show which geometry was actually evaluated.
108*/
109function demoPrintMetrics(): void {
110 global $demoMetrics;
111 echo "\nMetrics\n";
112 foreach ($demoMetrics['timings'] as $label => $seconds) {
113 printf(" %-35s %9.2f s\n", $label, $seconds);
114 }

116 foreach (['download_bytes' => 'Compressed download', 'extracted_bytes' => 'Extracted GeoJSON',
117 'database_bytes' => 'Stored polygon payload',
118 'load_peak_bytes' => $demoMetrics['phase_memory_available'] ? 'PHP peak: load + source capture' : 'PHP lifetime peak after loading',
119 'reference_peak_bytes' => $demoMetrics['phase_memory_available'] ? 'PHP peak: independent validation' : 'PHP lifetime peak after validation'] as $field => $label) {
120 if (isset($demoMetrics[$field])) { printf(" %-35s %9.2f MiB\n", $label, $demoMetrics[$field] / 1048576); }
121 }
122 if (!empty($demoMetrics['lookup_ms'])) {
123 echo ' All lookup calls: '.demoLatency($demoMetrics['lookup_ms'])."\n";
124 echo ' Known-location calls: '.demoLatency($demoMetrics['known_lookup_ms'])."\n";
125 echo ' Largest-shape calls: '.demoLatency($demoMetrics['generated_lookup_ms'])."\n";
126 printf(" Lookup throughput (API time only): %.1f calls/s\n", count($demoMetrics['lookup_ms']) * 1000 / array_sum($demoMetrics['lookup_ms']));
127 if ($demoMetrics['phase_memory_available']) {
128 printf(" Peak extra PHP memory per lookup: %.2f MiB (total peak %.2f MiB)\n", $demoMetrics['lookup_extra_bytes'] / 1048576, $demoMetrics['lookup_peak_bytes'] / 1048576);
129 } else {
130 echo " Per-phase memory measurement requires PHP 8.2 or later.\n";
131 }
132 }
133 if (isset($demoMetrics['largest'])) {
134 if (isset($demoMetrics['polygon_rows'])) {
135 printf(" Actual polygon rows evaluated: %d (%.2f MiB processed across lookup calls)\n", $demoMetrics['polygon_rows'], $demoMetrics['polygon_bytes_processed'] / 1048576);
136 printf(" Largest polygons decoded in generated tests: %d/%d\n", $demoMetrics['largest_decoded'], count($demoMetrics['largest']));
137 }
138 if (isset($demoMetrics['boundary_pairs'])) {
139 printf(" Boundary pairs with distinct source zones: %d/%d (%d named-to-named)\n", $demoMetrics['crossing_pairs'], $demoMetrics['boundary_pairs'], $demoMetrics['named_pairs']);
140 }
141 echo " Largest polygons tested (ranked by vertex count):\n";
142 foreach ($demoMetrics['largest'] as $shape) {
143 printf(" %-28s id %-5d %8d vertices %6.2f MiB\n", $shape['tzname'], $shape['id'], $shape['vertices'], $shape['bytes'] / 1048576);
144 }
145 }
146 if (isset($demoMetrics['lookup_ms']) && !empty($demoMetrics['lookup_ms'])) {
147 echo " Reference geometry is freed before measuring package lookup memory.\n";
148 }
149 echo " PHP memory describes this worker; the database server and Composer run separately.\n\n";
150}

274/**
275 \brief This removes the downloads and drops the database owned by this demo run.
276
277 The database name comes from database.json, and must match the generated demo naming pattern. We delete large
278 files before connecting to the database server, so they are removed even when the server is unavailable. If dropping the
279 database fails, its cleanup record remains for recovery. The shell removes the working directory only on success.
280
281 \returns: True, if cleanup succeeded, or no database had been created.
282 \throws RuntimeException if a cleanup record is unexpected, or a file cannot be removed.
283 \throws PDOException if database cleanup fails.
284*/
285function demoCleanup(string $directory ///< The working directory owned by this demo run.
286 ): bool {
287 $start = hrtime(true);
288 // Delete the large files even if the database server is temporarily unavailable during cleanup.
289 demoRemoveDownloads($directory);
290 $state = $directory.'/database.json';
291 if (!is_file($state)) {
292 printf("Cleanup: no temporary database to remove (%.3f s).\n", (hrtime(true) - $start) / 1e9);
293 return true;
294 }
295 $record = json_decode(file_get_contents($state), true, 512, JSON_THROW_ON_ERROR);
296 $database = $record['database'];
297 if (!preg_match('/^lgv_tz_demo_[a-f0-9]{24}$/D', $database)) {
298 throw new RuntimeException('Unexpected demo database name; refusing to drop it.');
299 }
300 echo 'Removing temporary database '.$database."...\n";
301 $driver = $record['driver'] ?? 'mysql';
302 if (!in_array($driver, ['mysql', 'pgsql'], true)) {
303 throw new RuntimeException('Unexpected recorded database driver; refusing to drop it.');
304 }
305 $quote = $driver === 'pgsql' ? '"' : '`';
306 demoConnection(null, $driver)->exec('DROP DATABASE IF EXISTS '.$quote.$database.$quote);
307 if (!unlink($state)) {
308 throw new RuntimeException('Could not remove the database cleanup record.');
309 }
310 printf("Database/download cleanup: %.3f s.\n", (hrtime(true) - $start) / 1e9);
311 return true;
312}

314/***************************************************************************************************************************/
315/**
316 \brief This creates a uniquely named database and records it for cleanup.
317
318 CREATE does not use IF NOT EXISTS, so a name collision cannot adopt an existing database. Where pcntl is
319 available, interruption signals are blocked until the cleanup record has been written. If the record cannot be
320 written, we drop the new database before reporting the failure.
321
322 \returns: The new database name, beginning with "lgv_tz_demo_".
323 \throws RuntimeException if the cleanup record cannot be written.
324 \throws PDOException if database creation or rollback fails.
325*/
326function demoCreateDatabase(PDO $admin, ///< An administrative connection with CREATE and DROP permissions.
327 string $directory ///< The working directory in which database.json is saved.
328 ): string {
329 $name = 'lgv_tz_demo_'.bin2hex(random_bytes(12));
330 $driver = $admin->getAttribute(PDO::ATTR_DRIVER_NAME);
331 $quote = $driver === 'pgsql' ? '"' : '`';
332 // Do not deliver an interruption between creating the database and saving its cleanup record.
333 $previousMask = [];
334 $blocked = function_exists('pcntl_sigprocmask') && pcntl_sigprocmask(SIG_BLOCK, [SIGHUP, SIGINT, SIGTERM], $previousMask);
335 try {
336 // CREATE without IF NOT EXISTS ensures this run can never adopt or reset an existing database.
337 $admin->exec('CREATE DATABASE '.$quote.$name.$quote);
338 if (false === file_put_contents($directory.'/database.json', json_encode(['database' => $name, 'driver' => $driver], JSON_THROW_ON_ERROR))) {
339 $admin->exec('DROP DATABASE '.$quote.$name.$quote);
340 throw new RuntimeException('Could not record the temporary database for cleanup.');
341 }
342 } finally {
343 if ($blocked) { pcntl_sigprocmask(SIG_SETMASK, $previousMask); }
344 }
345 return $name;
346}

349/**
350 \brief This performs the complete Composer installation, data load, system test, and metrics run.
351
352 We first check PHP and the database server, then install the package and obtain the latest boundary release. The package
353 performs the actual load while a listener retains independent source geometry for storage and lookup checks.
354 The large files are deleted after loading, and the reference geometry is freed before lookup measurements.
355
356 The known locations come from the installed package. Generated tests sample the largest polygons, including
357 points on both sides of boundaries. Coverage must show that those polygons were actually decoded by the API.
358 The parent shell handles database and directory cleanup after this function returns or the worker fails.
359
360 \returns: 0, if every check passes; 1, if a storage, lookup, or coverage check fails.
361 \throws RuntimeException if setup, download, loading, or test preparation cannot complete.
362*/
363function demoRun(string $directory ///< The private working directory created by run.sh.
364 ): int {
365 global $demoMetrics;
366 if (PHP_VERSION_ID < 80000) {
367 throw new RuntimeException('PHP 8.0 or later is required.');
368 }
369 $driver = demoDriver();
370 foreach (['pdo_'.$driver, 'curl', 'zip', 'mbstring', 'ctype'] as $extension) {
371 if (!extension_loaded($extension)) {
372 throw new RuntimeException('The PHP '.$extension.' extension is required.');
373 }
374 }
375 if (!function_exists('proc_open')) {
376 throw new RuntimeException('PHP proc_open must be enabled to run Composer.');
377 }
378 // Check credentials before downloading anything. The connection must be able to create and drop a database.
379 $admin = demoConnection();
380 printf("Database: %s %s.\n", $driver, $admin->getAttribute(PDO::ATTR_SERVER_VERSION));
381 $name = demoCreateDatabase($admin, $directory);
382 $start = hrtime(true);
383 demoInstall($directory);
384 demoTiming('Composer setup', $start);
385
386 echo "Finding the latest timezone boundary release...\n";
387 $start = hrtime(true);
388 demoDownload(DEMO_RELEASE_API, $directory.'/release.json');
389 $release = json_decode(file_get_contents($directory.'/release.json'), true, 512, JSON_THROW_ON_ERROR);
390 $asset = null;
391 foreach ($release['assets'] ?? [] as $candidate) {
392 if ($candidate['name'] === 'timezones-with-oceans.geojson.zip') {
393 $asset = $candidate;
394 break;
395 }
396 }
397 if (null === $asset) {
398 throw new RuntimeException('The latest release does not contain timezones-with-oceans.geojson.zip.');
399 }
400 printf("Downloading boundary release %s (%.1f MiB compressed)...\n", $release['tag_name'], $asset['size'] / 1048576);
401 $archive = $directory.'/boundaries.zip';
402 $boundary = $directory.'/boundaries.json';
403 demoDownload($asset['browser_download_url'], $archive);
404 if (isset($asset['digest']) && str_starts_with($asset['digest'], 'sha256:') &&
405 !hash_equals(substr($asset['digest'], 7), hash_file('sha256', $archive))) {
406 throw new RuntimeException('Boundary archive checksum did not match the release metadata.');
407 }
408 $demoMetrics['download_bytes'] = filesize($archive);
409 demoTiming('Release discovery + download', $start);
410 $start = hrtime(true);
411 demoExtract($archive, $boundary);
412 $demoMetrics['extracted_bytes'] = filesize($boundary);
413 demoTiming('ZIP extraction', $start);
414 printf("Extracted %.1f MiB of GeoJSON.\n", filesize($boundary) / 1048576);
415
416 echo 'Loading boundaries into temporary database '.$name."...\n";
417 require_once __DIR__.'/SystemTests.php';
418 $database = new DemoTrackedDatabase($name,
419 demoEnvironment('USER', $driver === 'pgsql' ? (getenv('USER') ?: get_current_user()) : 'root'),
420 demoEnvironment('PASSWORD', ''), $driver, demoEnvironment('HOST', 'localhost'),
421 (int)demoEnvironment('PORT', $driver === 'pgsql' ? '5432' : '3306'));
422 $stream = fopen($boundary, 'rb');
423 if (false === $stream) {
424 throw new RuntimeException('Could not open the boundary file.');
425 }
426 $start = hrtime(true);
427 demoResetMemoryPeak();
428 $source = new DemoSourceGeometry();
429 try {
430 $listener = new LGV_TZ_Lookup_Loader($database);
431 $inspector = new DemoInspectingListener($listener, $source);
432 $parser = new JsonStreamingParser\Parser($stream, $inspector);
433 $parser->parse();
434 $demoMetrics['load_peak_bytes'] = memory_get_peak_usage();
435 } finally {
436 fclose($stream);
437 unset($parser, $listener, $inspector);
438 demoRemoveDownloads($directory);
439 }
440 demoTiming('Load + capture reference geometry', $start);
441 $validation = demoConnection($name);
442 if ($driver === 'pgsql') {
443 // Give PostgreSQL current statistics before its first bounding-box queries.
444 $validation->exec('ANALYZE timezones');
445 }
446 $polygons = $validation->query('SELECT COUNT(*) FROM timezones')->fetchColumn();
447 printf("Loaded %s polygons in %.2f seconds. Download and extracted file deleted.\n", $polygons, (hrtime(true) - $start) / 1e9);
448 if ((int)$polygons === 0) {
449 throw new RuntimeException('No polygons were loaded.');
450 }
451
452 $start = hrtime(true);
453 demoResetMemoryPeak();
454 echo "Checking stored polygons against the decoded source GeoJSON...\n";
455 $storageFailures = $source->validateStorage($validation);
456 foreach ($storageFailures as $failure) { echo 'FAIL '.$failure."\n"; }
457 printf("Source/storage validation: %d polygons checked, %d failures.\n", count($source->shapes), count($storageFailures));
458 $demoMetrics['database_bytes'] = $source->polygonBytes;
459 $largestCount = filter_var(demoEnvironment('LARGEST_SHAPES', '10'), FILTER_VALIDATE_INT,
460 ['options' => ['min_range' => 1, 'max_range' => 100]]);
461 if (false === $largestCount) { throw new RuntimeException('LGV_TZ_DEMO_LARGEST_SHAPES must be between 1 and 100.'); }
462 $demoMetrics['largest'] = array_map(static fn($shape) => array_intersect_key($shape, array_flip(['id', 'tzname', 'vertices', 'bytes'])), $source->largest($largestCount));
463 echo 'Generating independent interior and two-sided boundary probes for the '.$largestCount." largest polygons...\n";
464 $generated = $source->casesForLargest($largestCount);
465 $pairs = [];
466 foreach ($generated as $case) {
467 if (isset($case['pair'])) { $pairs[$case['pair']][$case['side']] = $case['expected']; }
468 }
469 $demoMetrics['boundary_pairs'] = count($pairs);
470 $demoMetrics['crossing_pairs'] = 0;
471 $demoMetrics['named_pairs'] = 0;
472 foreach ($pairs as $sides) {
473 if (isset($sides[-1], $sides[1]) && $sides[-1] !== $sides[1]) {
474 ++$demoMetrics['crossing_pairs'];
475 $zones = array_merge($sides[-1], $sides[1]);
476 if (!empty($sides[-1]) && !empty($sides[1]) &&
477 count(array_filter($zones, static fn($z) => str_starts_with($z, 'Etc/'))) === 0) {
478 ++$demoMetrics['named_pairs'];
479 }
480 }
481 }
482 $demoMetrics['reference_peak_bytes'] = memory_get_peak_usage();
483 unset($source);
484 gc_collect_cycles();
485 if (function_exists('gc_mem_caches')) { gc_mem_caches(); }
486 demoTiming('Independent source/storage validation', $start);
487
488 $package = Composer\InstalledVersions::getInstallPath(DEMO_PACKAGE);
489 require $package.'/src/TestLocations.php';
490 if (empty($test_locations_param_array)) {
491 throw new RuntimeException('The installed package contains no location tests.');
492 }
493 $lookup = new LGV_TZ_Lookup_Query($database);
494 $cases = [];
495 foreach ($test_locations_param_array as $test) {
496 $test['expected'] = [$test['result']];
497 $test['kind'] = 'known location';
498 $cases[] = $test;
499 }
500 $cases = array_merge($cases, $generated);
501 $failures = count($storageFailures);
502 $groups = [];
503 $start = hrtime(true);
504 echo 'Running '.count($test_locations_param_array).' known locations and '.count($generated)." generated system tests through the Composer lookup API...\n";
505 foreach ($cases as $index => $test) {
506 $lng = $test['params']['lng'];
507 $lat = $test['params']['lat'];
508 while ($lng < -180) { $lng += 360; }
509 while ($lng > 180) { $lng -= 360; }
510 while ($lat < -90) { $lat += 180; }
511 while ($lat > 90) { $lat -= 180; }
512 demoResetMemoryPeak();
513 $baselineMemory = memory_get_usage();
514 $database->generatedCall = $test['kind'] !== 'known location';
515 $callStart = hrtime(true);
516 $actual = $lookup->get_tz($lng, $lat);
517 $elapsed = (hrtime(true) - $callStart) / 1e6;
518 $demoMetrics['lookup_ms'][] = $elapsed;
519 if ($test['kind'] !== 'known location') { $demoMetrics['generated_lookup_ms'][] = $elapsed; }
520 else { $demoMetrics['known_lookup_ms'][] = $elapsed; }
521 $demoMetrics['lookup_extra_bytes'] = max($demoMetrics['lookup_extra_bytes'], memory_get_peak_usage() - $baselineMemory);
