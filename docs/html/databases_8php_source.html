51/**
52 \brief This observes native PostgreSQL result sizes without keeping statements or cursors alive.
53 PHP heap counters alone do not include libpq's result buffers. A weak reference lets the test inspect the
54 active result and verify that it contains one row, while normal statement destruction still performs cleanup.
55*/
56class DatabaseTestStatement extends PDOStatement {
57 public static ?WeakReference $last = null; ///< The most recently prepared statement, while it remains alive.
58 protected function __construct() { self::$last = WeakReference::create($this); }
59}

103 // A known circle crosses two decoder blocks. A second named candidate forces actual polygon decoding.
104 $ring = [];
105 for ($i = 0; $i < 2053; ++$i) {
106 $angle = 2 * M_PI * $i / 2053;
107 $ring[] = [40 + cos($angle), 40 + sin($angle)];

113 foreach ([-0.99, -0.7, 0.0, 0.7, 0.99] as $x) {
114 foreach ([-0.99, -0.7, 0.0, 0.7, 0.99] as $y) {
115 databaseCheck($x * $x + $y * $y < 1 ? 'Region/Circle' : '', $lookup->get_tz(40 + $x, 40 + $y), 'Analytic circle '.$x.','.$y);
116 }
117 }
118 $sourceBytes = '';

120 $row = iterator_to_array($database->get_tz_polygons([6]))[0];

122 unset($ring, $row, $sourceBytes);

124 // Many one-MiB blobs would exceed this bound if the PDO reader buffered the whole result in PHP memory.
125 $type = $driver === 'pgsql' ? 'BYTEA' : 'LONGBLOB';
126 $wrapper->preparedStatement('CREATE TEMPORARY TABLE binary_fixture (id INTEGER PRIMARY KEY, payload '.$type.')');
127 $blob = str_repeat(implode('', array_map('chr', range(0, 255))), 4096);
128 for ($i = 1; $i <= 24; ++$i) {
129 $wrapper->preparedStatement('INSERT INTO binary_fixture VALUES (?, ?)', [$i, $blob], false, [1 => PDO::PARAM_LOB]);
130 }
131 $wrapper->preparedStatement('INSERT INTO binary_fixture VALUES (?, ?)', [25, ''], false, [1 => PDO::PARAM_LOB]);
132 databaseCheck('', $wrapper->preparedStatement('SELECT payload FROM binary_fixture WHERE id = 25', [], true)[0]['payload'], 'Empty binary value');
133 if (function_exists('memory_reset_peak_usage')) { memory_reset_peak_usage(); }
134 $before = memory_get_usage();
135 $count = 0;
136 $nativePeak = 0;

139 if ($driver === 'pgsql' && defined('Pdo\\Pgsql::ATTR_RESULT_MEMORY_SIZE')) {

141 }

143 }
144 databaseCheck(24, $count, 'All binary rows fetched');
145 if (function_exists('memory_reset_peak_usage')) {

147 databaseCheck(true, $extra < 8 * 1048576, 'Binary reader stays below eight MiB extra PHP memory');
148 printf(" Binary reader peak extra PHP memory: %.2f MiB for 24 MiB of rows.\n", $extra / 1048576);
149 }
150 if ($nativePeak > 0) {
151 databaseCheck(true, $nativePeak < 4 * 1048576, 'PostgreSQL native result buffer holds one row');
